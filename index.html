<script>

    // Ganti CLIENT_ID dengan Client ID resmi dari pengembang Strava kamu (jika sudah daftar di strava.com/settings/api)

    const CLIENT_ID = 'GANTI_DENGAN_STRAVA_CLIENT_ID'; 

    

    function connectStravaReal() {

        if(CLIENT_ID === 'GANTI_DENGAN_STRAVA_CLIENT_ID') {

            alert("Catatan: Untuk koneksi asli, daftarkan aplikasi di Strava API Settings lalu masukkan Client ID. Untuk sekarang, menggunakan mode simulasi aktif.");

            document.getElementById("stravaStatus").innerText = "Status: Terhubung (Mode Simulasi Siap)";

            return;

        }

        const redirectUri = window.location.href.split('?')[0];

        const authUrl = `https://www.strava.com/oauth/authorize?client_id=${CLIENT_ID}&response_type=code&redirect_uri=${redirectUri}&approval_prompt=force&scope=read,activity:read_all`;

        window.location.href = authUrl;

    }


    function getAIWorkout() {

        const workouts = [

            "Hari ini: Easy Run santai sejauh 6 KM. Jaga Heart Rate di Zona 2 (Pace nyaman, masih bisa ngobrol).",

            "Hari ini: Interval Training. 1 km Warm-up, 4x (400 meter cepat + 400 meter jog), lalu 1 km Cool-down.",

            "Hari ini: Tempo Run. 2 km Easy, 3 km Pace Target Race, 1 km Cool-down."

        ];

        document.getElementById("todayWorkout").innerText = workouts[Math.floor(Math.random() * workouts.length)];

    }


    function askCoach() {

        const question = document.getElementById("userQuestion").value;

        const replyBox = document.getElementById("coachReply");

        if(!question) {

            alert("Tuliskan pertanyaanmu terlebih dahulu!");

            return;

        }

        replyBox.style.display = "block";

        replyBox.innerText = "Coach sedang menganalisis...";

        setTimeout(() => {

            replyBox.innerText = `Jawaban Coach untuk "${question}": Pastikan pemanasan dinamis 10 menit sebelum mulai, dan jaga hidrasi yang cukup!`;

        }, 800);

    }
